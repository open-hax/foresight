func TestCapacityHTTPMissingAuthAndNoLocal(t *testing.T) {
	h := newCapacityHTTPFixture(t)
	const body = `{"model":"gpt-test","session_id":"session-1","messages":[{"role":"user","content":"safe request"}]}`
	request := func(token string) *httptest.ResponseRecorder {
		r := httptest.NewRequest(http.MethodPost, "/v1/chat/completions", strings.NewReader(body))
		if token != "" {
			r.Header.Set("Authorization", "Bearer "+token)
		}
		r.Header.Set("Idempotency-Key", "capacity-http-denial")
		w := httptest.NewRecorder()
		h.server.Handler.ServeHTTP(w, r)
		return w
	}
	unauthorized := request("")
	if unauthorized.Code != http.StatusUnauthorized && unauthorized.Code != http.StatusForbidden {
		t.Fatalf("missing auth status = %d", unauthorized.Code)
	}
	w := request(h.operatorToken)
	if w.Code != http.StatusServiceUnavailable {
		t.Fatalf("authorized status = %d body=%s", w.Code, w.Body.String())
	}
	var envelope map[string]json.RawMessage
	if err := json.Unmarshal(w.Body.Bytes(), &envelope); err != nil {
		t.Fatal(err)
	}
	if len(envelope) != 1 || envelope["error"] == nil {
		t.Fatalf("unexpected envelope keys: %v", envelope)
	}
	var detail map[string]json.RawMessage
	if err := json.Unmarshal(envelope["error"], &detail); err != nil {
		t.Fatal(err)
	}
	if len(detail) != 5 {
		t.Fatalf("unexpected error keys: %v", detail)
	}
	for _, key := range []string{"code", "message", "details", "trace_id", "timestamp"} {
		if detail[key] == nil {
			t.Fatalf("missing error key %s", key)
		}
	}
	var code, message, trace, timestamp string
	_ = json.Unmarshal(detail["code"], &code)
	_ = json.Unmarshal(detail["message"], &message)
	_ = json.Unmarshal(detail["trace_id"], &trace)
	_ = json.Unmarshal(detail["timestamp"], &timestamp)
	if code != goverror.CodeInferenceProcessingFailed || message != "failed to process request" || trace == "" {
		t.Fatalf("unexpected governed error: code=%s message=%s trace=%s", code, message, trace)
	}
	if _, err := time.Parse(time.RFC3339Nano, timestamp); err != nil {
		t.Fatalf("invalid timestamp: %v", err)
	}
	var details map[string]any
	if err := json.Unmarshal(detail["details"], &details); err != nil {
		t.Fatal(err)
	}
	if len(details) != 1 || details["reason"] != "capacity_unavailable" {
		t.Fatalf("unexpected public details: %v", details)
	}
}
